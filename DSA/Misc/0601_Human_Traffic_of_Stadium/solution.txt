# 0601. Human Traffic of Stadium
# Difficulty : Hard
# Tags       : database
# Solved on  : 2020-01-08
# LeetCode   : https://leetcode.com/problems/human-traffic-of-stadium/

# Write your MySQL query statement below
select t1.* from stadium t1,stadium t2,stadium t3 
where t1.people>=100 and t2.people>=100 and t3.people>=100
and (
    t1.id-t2.id=1 and t1.id-t3.id=2 and t2.id-t3.id=1
    or 
    t2.id-t1.id=1 and t2.id-t3.id=2 and t1.id-t3.id=1
    or
    t3.id-t2.id=1 and t3.id-t1.id=2 and t2.id-t1.id=1
)
group by t1.id
