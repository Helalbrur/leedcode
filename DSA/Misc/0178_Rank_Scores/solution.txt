# 0178. Rank Scores
# Difficulty : Medium
# Tags       : database
# Solved on  : 2018-11-27
# LeetCode   : https://leetcode.com/problems/rank-scores/

# Write your MySQL query statement below
select Score,(select count(distinct(Score)) from Scores b where b.Score>=a.Score) as Rank from Scores a order by Score DESC ;
